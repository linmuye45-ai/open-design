name: Koi Fortune CI

# 仅在 koi/ 变更时运行，测试 + 构建原型
on:
  push:
    paths:
      - "koi/**"
      - ".github/workflows/koi-ci.yml"
  pull_request:
    paths:
      - "koi/**"
      - ".github/workflows/koi-ci.yml"
  workflow_dispatch:

jobs:
  test-build:
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: koi
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: "20"
      - name: Install
        run: npm install
      - name: Test
        run: npm test
      - name: Build
        run: npm run build
